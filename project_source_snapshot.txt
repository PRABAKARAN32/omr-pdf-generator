class AppShell extends ConsumerStatefulWidget {
  const AppShell({super.key});

  @override
  ConsumerState<AppShell> createState() => _AppShellState();
}

class _AppShellState extends ConsumerState<AppShell> {
  int selectedIndex = 0;

  static const List<String> pageTitles = [
    'Dashboard',
    'New Job',
    'Excel Import',
    'Barcode Configuration',
    'PDF Layout Designer',
    'PDF Preview',
  ];

  void _goToPage(int index) {
    if (index < 0 || index >= pageTitles.length) return;

    setState(() {
      selectedIndex = index;
    });
  }

  void _startNewProject() {
    ref.read(jobSessionProvider.notifier).clearSession();
    ref.read(excelProvider.notifier).reset();
    ref.read(barcodeConfigurationProvider.notifier).reset();
    ref.read(layoutConfigurationProvider.notifier).reset();

    _goToPage(1);
  }

  Widget _buildPlaceholderPage({
    required String title,
    required String description,
    required VoidCallback onBack,
  }) {
    return Center(
      child: Padding(
        padding: const EdgeInsets.all(32),
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            Text(title, style: Theme.of(context).textTheme.headlineMedium),
            const SizedBox(height: 12),
            Text(description, textAlign: TextAlign.center),
            const SizedBox(height: 24),
            ElevatedButton.icon(
              onPressed: onBack,
              icon: const Icon(Icons.arrow_back),
              label: const Text('Back'),
            ),
          ],
        ),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final pages = <Widget>[
      // 0 - Dashboard
      DashboardScreen(onNewProject: _startNewProject),

      // 1 - New Job
      NewJobScreen(onBack: () => _goToPage(0), onContinue: () => _goToPage(2)),

      // 2 - Excel Import
      ExcelImportScreen(onBack: () => _goToPage(1), onNext: () => _goToPage(3)),

      // 3 - Barcode Configuration
      BarcodeConfigurationScreen(
        onBack: () => _goToPage(2),
        onNext: () => _goToPage(4),
      ),

      // 4 - PDF Layout Designer
      LayoutDesignerScreen(
        onBack: () => _goToPage(3),

      // 5 - PDF Preview
      _buildPlaceholderPage(
        title: 'PDF Preview',

class BarcodeConfiguration {

  BarcodeConfiguration copyWith({

import 'dart:math' as math;

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../excel_import/providers/excel_provider.dart';
import '../models/barcode_configuration.dart';
import '../providers/barcode_configuration_provider.dart';

class BarcodeConfigurationScreen extends ConsumerStatefulWidget {
  final VoidCallback onBack;
  final VoidCallback onNext;

  const BarcodeConfigurationScreen({
    super.key,
    required this.onBack,
    required this.onNext,
  });

  @override
  ConsumerState<BarcodeConfigurationScreen> createState() =>
      _BarcodeConfigurationScreenState();
}

class _BarcodeConfigurationScreenState
    extends ConsumerState<BarcodeConfigurationScreen> {
  late final TextEditingController _startingNumberController;
  late final ProviderSubscription<BarcodeConfiguration> _subscription;

  @override
  void initState() {
    super.initState();

    final initialConfiguration = ref.read(barcodeConfigurationProvider);

    _startingNumberController = TextEditingController(
      text: initialConfiguration.startingNumber,
    );

    _subscription = ref.listenManual<BarcodeConfiguration>(
      barcodeConfigurationProvider,
      (previous, next) {
        if (_startingNumberController.text != next.startingNumber) {
          _startingNumberController.value = TextEditingValue(
            text: next.startingNumber,
            selection: TextSelection.collapsed(
              offset: next.startingNumber.length,
            ),
          );
        }
      },
    );
  }

  @override
  void dispose() {
    _subscription.close();
    _startingNumberController.dispose();
    super.dispose();
  }

  String _formatNumber(BigInt number, int width) {
    return number.toString().padLeft(width, '0');
  }

  String? _validateConfiguration({
    required BarcodeFormat format,
    required String startingNumber,
    required int recordCount,
  }) {
    if (recordCount <= 0) {
      return 'No candidate records are available. Please import Excel data.';
    }

    if (startingNumber.isEmpty) {
      return 'Starting barcode number is required.';
    }

    if (!RegExp(r'^[0-9]+$').hasMatch(startingNumber)) {
      return 'Only numeric starting numbers are supported.';
    }

    if (format == BarcodeFormat.code32 && startingNumber.length > 8) {
      return 'Code 32 supports a maximum of 8 input digits.';
    }

    if (startingNumber.length > 18) {
      return 'Starting number cannot exceed 18 digits.';
    }

    final start = BigInt.tryParse(startingNumber);

    if (start == null) {
      return 'Please enter a valid starting number.';
    }

    final last = start + BigInt.from(recordCount - 1);

    if (format == BarcodeFormat.code32 && last > BigInt.from(99999999)) {
      return 'The generated Code 32 sequence exceeds its 8-digit range.';
    }

    if (last.toString().length > 18) {
      return 'The generated barcode sequence exceeds the supported range.';
    }

    return null;
  }

  void _continue() {
    final configuration = ref.read(barcodeConfigurationProvider);
    final excelData = ref.read(excelProvider).importedData;

    if (excelData == null) {
      _showMessage('Please import Excel data first.');
      return;
    }

    final error = _validateConfiguration(
      format: configuration.format,
      startingNumber: configuration.startingNumber,
      recordCount: excelData.recordCount,
    );

    if (error != null) {
      _showMessage(error);
      return;
    }

    widget.onNext();
  }

  void _showMessage(String message) {
    if (!mounted) return;

    ScaffoldMessenger.of(context)
        .showSnackBar(SnackBar(content: Text(message)));
  }

  Widget _buildFormatOption({
    required BarcodeFormat format,
    required String description,
    required BarcodeFormat selectedFormat,
    required ValueChanged<BarcodeFormat> onSelected,
  }) {
    final isSelected = selectedFormat == format;

    return InkWell(
      onTap: () => onSelected(format),
      borderRadius: BorderRadius.circular(6),
      child: Container(
        width: 220,
        padding: const EdgeInsets.all(14),
        decoration: BoxDecoration(
          border: Border.all(
            color: isSelected
                ? Theme.of(context).colorScheme.primary
                : Colors.grey.shade400,
            width: isSelected ? 2 : 1,
          ),
          borderRadius: BorderRadius.circular(6),
        ),
        child: Row(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Radio<BarcodeFormat>(
              value: format,
              groupValue: selectedFormat,
              onChanged: (value) {
                if (value != null) onSelected(value);
              },
            ),
            const SizedBox(width: 8),
            Expanded(
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    format.label,
                    style: const TextStyle(fontWeight: FontWeight.bold),
                  ),
                  const SizedBox(height: 5),
                  Text(
                    description,
                    style: const TextStyle(fontSize: 12, color: Colors.grey),
                  ),
                ],
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _buildSequencePreview({
    required BarcodeConfiguration configuration,
    required int recordCount,
  }) {
    final startText = configuration.startingNumber;

    if (startText.isEmpty ||
        !RegExp(r'^[0-9]+$').hasMatch(startText) ||
        recordCount <= 0) {
      return const SizedBox.shrink();
    }

    final start = BigInt.tryParse(startText);

    if (start == null) {
      return const SizedBox.shrink();
    }

    final width = configuration.format == BarcodeFormat.code32
        ? 8
        : startText.length;

    final normalizedStart = BigInt.parse(startText.padLeft(width, '0'));

    final last = normalizedStart + BigInt.from(recordCount - 1);

    if (configuration.format == BarcodeFormat.code32 &&
        last > BigInt.from(99999999)) {
      return const SizedBox.shrink();
    }

    if (last.toString().length > 18) {
      return const SizedBox.shrink();
    }

    final firstValue = _formatNumber(normalizedStart, width);
    final lastValue = _formatNumber(last, width);

    return Container(
      width: double.infinity,
      padding: const EdgeInsets.all(16),
      decoration: BoxDecoration(
        border: Border.all(color: Colors.grey.shade400),
        borderRadius: BorderRadius.circular(6),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          const Text(
            'Barcode Sequence Preview',
            style: TextStyle(fontWeight: FontWeight.bold),
          ),
          const SizedBox(height: 14),
          Text('First barcode: $firstValue'),
          const SizedBox(height: 6),
          Text('Last barcode: $lastValue'),
          const SizedBox(height: 6),
          Text('Total barcodes: $recordCount'),
          if (configuration.format == BarcodeFormat.code32) ...[
            const SizedBox(height: 8),
            const Text(
              'Code 32 will use an 8-digit numeric input and calculate '
              'its required check digit during barcode encoding.',
              style: TextStyle(fontSize: 12, color: Colors.grey),
            ),
          ],
        ],
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final configuration = ref.watch(barcodeConfigurationProvider);
    final excelData = ref.watch(excelProvider).importedData;
    final recordCount = excelData?.recordCount ?? 0;

    final notifier = ref.read(barcodeConfigurationProvider.notifier);

    final descriptions = {
      BarcodeFormat.code32:
          'Numeric format with an 8-digit data value and check digit.',
      BarcodeFormat.code39: 'Standard alphanumeric barcode format.',
      BarcodeFormat.code128:
          'Compact barcode format for numeric and alphanumeric data.',
    };

    final validationError = _validateConfiguration(
      format: configuration.format,
      startingNumber: configuration.startingNumber,
      recordCount: recordCount,
    );

    return Padding(
      padding: const EdgeInsets.all(24),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          const Row(
            children: [
              Text(
                'Barcode Configuration',
                style: TextStyle(fontSize: 24, fontWeight: FontWeight.bold),
              ),
            ],
          ),

          const SizedBox(height: 8),

          Text(
            'Choose a barcode format and define the starting number.',
            style: TextStyle(color: Colors.grey.shade700),
          ),

          const SizedBox(height: 28),

          const Text(
            '1. Select Barcode Algorithm',
            style: TextStyle(fontWeight: FontWeight.bold),
          ),

          const SizedBox(height: 12),

          Wrap(
            spacing: 12,
            runSpacing: 12,
            children: BarcodeFormat.values.map((format) {
              return _buildFormatOption(
                format: format,
                description: descriptions[format]!,
                selectedFormat: configuration.format,
                onSelected: notifier.updateFormat,
              );
            }).toList(),
          ),

          const SizedBox(height: 28),

          const Text(
            '2. Starting Barcode Number',
            style: TextStyle(fontWeight: FontWeight.bold),
          ),

          const SizedBox(height: 10),

          SizedBox(
            width: 350,
            child: TextField(
              controller: _startingNumberController,
              keyboardType: TextInputType.number,
              inputFormatters: [
                FilteringTextInputFormatter.digitsOnly,
                LengthLimitingTextInputFormatter(18),
              ],
              decoration: InputDecoration(
                hintText: 'Example: 50001',
                border: const OutlineInputBorder(),
                helperText: configuration.format == BarcodeFormat.code32
                    ? 'Code 32 accepts up to 8 input digits.'
                    : 'Numeric sequence starting value.',
              ),
              onChanged: notifier.updateStartingNumber,
            ),
          ),

          const SizedBox(height: 24),

          Row(
            children: [
              const Text(
                'Candidate records:',
                style: TextStyle(fontWeight: FontWeight.bold),
              ),
              const SizedBox(width: 8),
              Text(recordCount.toString()),
            ],
          ),

          const SizedBox(height: 16),

          _buildSequencePreview(
            configuration: configuration,
            recordCount: recordCount,
          ),

          if (validationError != null) ...[
            const SizedBox(height: 12),
            Text(validationError, style: const TextStyle(color: Colors.red)),
          ],

          const Spacer(),

          Row(
            children: [
              OutlinedButton.icon(
                onPressed: widget.onBack,
                icon: const Icon(Icons.arrow_back),
                label: const Text('Back'),
              ),
              const Spacer(),
              ElevatedButton.icon(
                onPressed: validationError == null ? _continue : null,
                icon: const Icon(Icons.arrow_forward),
                label: const Text('Next: Layout Designer'),
              ),
            ],
          ),
        ],
      ),
    );
  }
}

      'jobId': jobId,

      clientName: json['clientName'] as String,
      orderDate: DateTime.parse(json['orderDate'] as String),
      createdAt: DateTime.parse(json['createdAt'] as String),

  }

final excelProvider = NotifierProvider<ExcelImportNotifier, ExcelImportState>(

class ExcelImportNotifier extends Notifier<ExcelImportState> {
  @override
  ExcelImportState build() {
    return const ExcelImportState();
  }

  void selectWorkbook({required String fileName, required Uint8List bytes}) {
    try {
      debugPrint('EXCEL: Starting workbook decoding');
      debugPrint('EXCEL: File = $fileName');
      debugPrint('EXCEL: Size = ${bytes.length} bytes');

      final service = ref.read(excelServiceProvider);
      final sheets = service.getWorksheetNames(bytes);

      debugPrint('EXCEL: Worksheets detected = ${sheets.length}');
      debugPrint('EXCEL: $sheets');

      if (sheets.isEmpty) {
        throw Exception('No worksheets found in this workbook.');
      }

      // Automatically import the first worksheet.
      final firstSheet = sheets.first;

      final data = service.importWorksheet(
        bytes: bytes,
        fileName: fileName,
        sheetName: firstSheet,
      );

      debugPrint('EXCEL: Imported worksheet = $firstSheet');
      debugPrint('EXCEL: Headers = ${data.headers.length}');
      debugPrint('EXCEL: Records = ${data.recordCount}');
      debugPrint('EXCEL: Warnings = ${data.warnings.length}');

      state = ExcelImportState(
        fileName: fileName,
        fileBytes: bytes,
        worksheetNames: sheets,
        importedData: data,
      );

      debugPrint('EXCEL: Workbook loaded successfully');
    } catch (error, stackTrace) {
      debugPrint('EXCEL ERROR: $error');
      debugPrintStack(stackTrace: stackTrace);

      state = ExcelImportState(
        fileName: fileName,
        fileBytes: bytes,
        error: error.toString(),
      );
    }
  }

  void toggleColumn(int columnIndex) {
    final data = state.importedData;

    if (data == null || columnIndex < 0 || columnIndex >= data.headers.length) {
      return;
    }

    final included = Set<int>.from(data.includedColumns);

    if (included.contains(columnIndex)) {
      included.remove(columnIndex);
    } else {
      included.add(columnIndex);
    }

    state = state.copyWith(
      importedData: data.copyWith(includedColumns: included),
    );
  }

  void selectAllColumns() {
    final data = state.importedData;
    if (data == null) return;

    state = state.copyWith(
      importedData: data.copyWith(
        includedColumns: Set<int>.from(
          List.generate(data.headers.length, (index) => index),
        ),
      ),
    );
  }

  void clearSelection() {
    final data = state.importedData;
    if (data == null) return;

    state = state.copyWith(
      importedData: data.copyWith(includedColumns: <int>{}),
    );
  }

  void reset() {
    state = const ExcelImportState();
  }
}

import 'dart:convert';
import 'dart:typed_data';

import 'package:archive/archive.dart';
import 'package:excel/excel.dart';

import '../models/excel_data.dart';

class ExcelService {
  Excel decodeWorkbook(Uint8List bytes) {
    final inputArchive = ZipDecoder().decodeBytes(bytes);
    final outputArchive = Archive();

    for (final file in inputArchive.files) {
      if (file.isFile && file.name == 'xl/_rels/workbook.xml.rels') {
        final xml = utf8.decode(file.content);

        final correctedXml = xml.replaceAllMapped(
          RegExp(r'Target="/xl/([^"]+)"'),
          (match) => 'Target="${match[1]}"',
        );

        final correctedBytes = utf8.encode(correctedXml);

        outputArchive.addFile(
          ArchiveFile(file.name, correctedBytes.length, correctedBytes),
        );
      } else {
        outputArchive.addFile(file);
      }
    }

    final correctedBytes = ZipEncoder().encode(outputArchive);

    if (correctedBytes == null) {
      throw Exception('Failed to normalize Excel workbook.');
    }

    return Excel.decodeBytes(correctedBytes);
  }

  List<String> getWorksheetNames(Uint8List bytes) {
    final workbook = decodeWorkbook(bytes);
    return workbook.tables.keys.toList();
  }

  ExcelImportData importWorksheet({
    required Uint8List bytes,
    required String fileName,
    required String sheetName,
    int headerRowIndex = 0,
  }) {
    final workbook = decodeWorkbook(bytes);
    final sheet = workbook.tables[sheetName];

    if (sheet == null) {
      throw Exception('Worksheet "$sheetName" was not found.');
    }

    final sourceRows = sheet.rows;

    if (sourceRows.isEmpty ||
        headerRowIndex < 0 ||
        headerRowIndex >= sourceRows.length) {
      throw Exception('The selected worksheet has no valid header row.');
    }

    // Read headers.
    final rawHeaders = sourceRows[headerRowIndex].map(_cellToText).toList();

    if (rawHeaders.every((header) => header.isEmpty)) {
      throw Exception('The selected header row is empty.');
    }

    // Generate unique column names.
    final headers = <String>[];
    final headerCounts = <String, int>{};

    for (var i = 0; i < rawHeaders.length; i++) {
      var header = rawHeaders[i];

      if (header.isEmpty) {
        header = 'Column ${i + 1}';
      }

      final count = (headerCounts[header] ?? 0) + 1;
      headerCounts[header] = count;

      if (count > 1) {
        header = '$header ($count)';
      }

      headers.add(header);
    }

    final importedRows = <List<String>>[];
    final warnings = <String>[];

    // Read candidate records after the header row.
    for (
      var rowIndex = headerRowIndex + 1;
      rowIndex < sourceRows.length;
      rowIndex++
    ) {
      final sourceRow = sourceRows[rowIndex];

      final row = List<String>.generate(headers.length, (columnIndex) {
        if (columnIndex >= sourceRow.length) {
          return '';
        }

        return _cellToText(sourceRow[columnIndex]);
      });

      // Ignore completely empty rows.
      if (row.every((value) => value.isEmpty)) {
        continue;
      }

      // Preserve partially filled rows and report missing values.
      final emptyColumns = <String>[];

      for (var i = 0; i < row.length; i++) {
        if (row[i].isEmpty) {
          emptyColumns.add(headers[i]);
        }
      }

      if (emptyColumns.isNotEmpty) {
        warnings.add(
          'Row ${rowIndex + 1}: Empty values in '
          '${emptyColumns.join(', ')}.',
        );
      }

      importedRows.add(row);
    }

    if (importedRows.isEmpty) {
      throw Exception('No candidate records found in this worksheet.');
    }

    return ExcelImportData(
      fileName: fileName,
      sheetName: sheetName,
      headers: headers,
      rows: importedRows,
      warnings: warnings,
      includedColumns: Set<int>.from(
        List.generate(headers.length, (index) => index),
      ),
    );
  }

  String _cellToText(Data? cell) {
    if (cell == null || cell.value == null) {
      return '';
    }

    return cell.value.toString().trim();
  }
}

class JobStorageService {
  Future<Directory> _getJobsDirectory() async {
    final appDirectory = await getApplicationSupportDirectory();

    final jobsDirectory = Directory(
      '${appDirectory.path}/OMR_PDF_Generator/jobs',
    );

    if (!await jobsDirectory.exists()) {
      await jobsDirectory.create(recursive: true);
    }

    return jobsDirectory;
  }

  Future<List<JobInfo>> getSavedJobs() async {
    final jobsDirectory = await _getJobsDirectory();

    final entities = await jobsDirectory.list(followLinks: false).toList();

    final List<JobInfo> jobs = [];

    for (final entity in entities) {
      if (entity is! Directory) continue;

      final infoFile = File('${entity.path}/job_info.json');

      if (!await infoFile.exists()) continue;

      final content = await infoFile.readAsString();

      final json = jsonDecode(content) as Map<String, dynamic>;

      jobs.add(JobInfo.fromJson(json));
    }

    jobs.sort((a, b) {
      final aDate = a.generatedAt ?? a.orderDate;
      final bDate = b.generatedAt ?? b.orderDate;

      return bDate.compareTo(aDate);
    });

    return jobs;
  }

  Future<void> deleteJob(String jobId) async {
    final jobsDirectory = await _getJobsDirectory();

    // Only allow a single directory name, not an arbitrary path.
    if (jobId.isEmpty ||
        jobId.contains('/') ||
        jobId.contains(r'\') ||
        jobId == '.' ||
        jobId == '..') {
      throw ArgumentError('Invalid job ID.');
    }

    final jobDirectory = Directory('${jobsDirectory.path}/$jobId');

    if (await jobDirectory.exists()) {
      await jobDirectory.delete(recursive: true);
    }
  }
}

          defaults.detailsFontSize,

          (json['fieldSpacing'] as num?)?.toDouble() ?? defaults.fieldSpacing,

          json['showBarcodeNumber'] as bool? ?? defaults.showBarcodeNumber,
      textStyle: LayoutTextStyle.values.firstWhere(
        (value) => value.name == json['textStyle'],
        orElse: () => defaults.textStyle,
      ),
      applyDetailsToAll:
          json['applyDetailsToAll'] as bool? ?? defaults.applyDetailsToAll,
      applyBarcodeToAll:
          json['applyBarcodeToAll'] as bool? ?? defaults.applyBarcodeToAll,
      barcodePageOverrides: overrides,

class LayoutConfigurationNotifier extends Notifier<LayoutConfiguration> {
  int _selectedPage = 1;

  int get selectedPage => _selectedPage;

  @override
  LayoutConfiguration build() {
    return const LayoutConfiguration();
  }

  void selectPage(int pageNumber) {
    if (pageNumber < 1) return;
    _selectedPage = pageNumber;
  }

  void reset() {
    _selectedPage = 1;
    state = const LayoutConfiguration();
  }

  void selectBox(LayoutBoxType type) {
    state = state.copyWith(selectedBox: type);
  }

  LayoutBox getBoxForPage(LayoutBoxType type, int pageNumber) {
    if (type == LayoutBoxType.details) {
      return state.detailsBox;
    }

    return state.barcodeBoxForPage(pageNumber);
  }

  void moveBox(
    LayoutBoxType type, {
    double dx = 0,
    double dy = 0,
    int? pageNumber,
  }) {
    final page = pageNumber ?? _selectedPage;
    final currentBox = getBoxForPage(type, page);

    final maxX = LayoutConfiguration.pageWidth - currentBox.width;
    final maxY = LayoutConfiguration.pageHeight - currentBox.height;

    final newX = (currentBox.x + dx).clamp(0.0, maxX).toDouble();
    final newY = (currentBox.y + dy).clamp(0.0, maxY).toDouble();

    final updatedBox = currentBox.copyWith(x: newX, y: newY);

    if (type == LayoutBoxType.details) {
      state = state.copyWith(detailsBox: updatedBox);
      return;
    }

    final overrides = Map<int, LayoutBox>.from(state.barcodePageOverrides);

    // Moving the barcode on a page creates an individual override.
    overrides[page] = updatedBox;

    state = state.copyWith(barcodePageOverrides: overrides);
  }

  void resizeBox(
    LayoutBoxType type, {
    double widthDelta = 0,
    double heightDelta = 0,
    int? pageNumber,
  }) {
    final page = pageNumber ?? _selectedPage;
    final currentBox = getBoxForPage(type, page);

    final maxWidth = LayoutConfiguration.pageWidth - currentBox.x;
    final maxHeight = LayoutConfiguration.pageHeight - currentBox.y;

    final newWidth = (currentBox.width + widthDelta)
        .clamp(1.0, maxWidth)
        .toDouble();

    final newHeight = (currentBox.height + heightDelta)
        .clamp(1.0, maxHeight)
        .toDouble();

    final updatedBox = currentBox.copyWith(width: newWidth, height: newHeight);

    if (type == LayoutBoxType.details) {
      state = state.copyWith(detailsBox: updatedBox);
      return;
    }

    final overrides = Map<int, LayoutBox>.from(state.barcodePageOverrides);

    overrides[page] = updatedBox;

    state = state.copyWith(barcodePageOverrides: overrides);
  }

  void updateSelectedBoxDimensions({double? width, double? height}) {
    final type = state.selectedBox;
    final currentBox = getBoxForPage(type, _selectedPage);

    final newWidth = (width ?? currentBox.width)
        .clamp(1.0, LayoutConfiguration.pageWidth - currentBox.x)
        .toDouble();

    final newHeight = (height ?? currentBox.height)
        .clamp(1.0, LayoutConfiguration.pageHeight - currentBox.y)
        .toDouble();

    final updatedBox = currentBox.copyWith(width: newWidth, height: newHeight);

    if (type == LayoutBoxType.details) {
      state = state.copyWith(detailsBox: updatedBox);
      return;
    }

    final overrides = Map<int, LayoutBox>.from(state.barcodePageOverrides);

    overrides[_selectedPage] = updatedBox;

    state = state.copyWith(barcodePageOverrides: overrides);
  }

  void resetBarcodePositionForPage(int pageNumber) {
    final overrides = Map<int, LayoutBox>.from(state.barcodePageOverrides);

    overrides.remove(pageNumber);

    state = state.copyWith(barcodePageOverrides: overrides);
  }

  void applyBarcodePositionToAllPages() {
    state = state.copyWith(barcodePageOverrides: const {});
  }

  void updateMovementStep(double value) {
    state = state.copyWith(movementStep: value.clamp(1.0, 20.0).toDouble());
  }

  void updateDetailsFontSize(double value) {
    state = state.copyWith(detailsFontSize: value);
  }

  void updateFieldSpacing(double value) {
    state = state.copyWith(fieldSpacing: value);
  }

  void updateShowBarcodeNumber(bool value) {
    state = state.copyWith(showBarcodeNumber: value);
  }

  void updateTextStyle(LayoutTextStyle value) {
    state = state.copyWith(textStyle: value);
  }

  void updateApplyDetailsToAll(bool value) {
    state = state.copyWith(applyDetailsToAll: value);
  }

  void updateApplyBarcodeToAll(bool value) {
    state = state.copyWith(applyBarcodeToAll: value);
  }
}

import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../excel_import/providers/excel_provider.dart';

class LayoutWarning {
  final int pageNumber;
  final String message;

  const LayoutWarning({required this.pageNumber, required this.message});
}

final layoutWarningsProvider = Provider<List<LayoutWarning>>((ref) {
  final configuration = ref.watch(layoutConfigurationProvider);
  final excelData = ref.watch(excelProvider).importedData;

  if (excelData == null || excelData.rows.isEmpty) {
    return const [];
  }

  final warnings = <LayoutWarning>[];

  final detailsBox = configuration.detailsBox;
  final fontSize = configuration.detailsFontSize;
  final fontWeight = configuration.textStyle == LayoutTextStyle.bold
      ? FontWeight.bold
      : FontWeight.normal;

  const padding = 12.0;

  final availableWidth = detailsBox.width - padding;
  final availableHeight = detailsBox.height - padding;

  final selectedColumns = excelData.includedColumns.toList()..sort();

  if (selectedColumns.isEmpty) {
    return const [
      LayoutWarning(
        pageNumber: 1,
        message: 'No fields selected for the Details Box.',
      ),
    ];
  }

  for (var rowIndex = 0; rowIndex < excelData.rows.length; rowIndex++) {
    final pageNumber = rowIndex + 1;
    final candidate = excelData.rows[rowIndex];

    final barcodeBox = configuration.barcodeBoxForPage(pageNumber);

    // Check physical intersection between the two boxes.
    final boxesOverlap =
        detailsBox.x < barcodeBox.x + barcodeBox.width &&
        detailsBox.x + detailsBox.width > barcodeBox.x &&
        detailsBox.y < barcodeBox.y + barcodeBox.height &&
        detailsBox.y + detailsBox.height > barcodeBox.y;

    if (boxesOverlap) {
      warnings.add(
        LayoutWarning(
          pageNumber: pageNumber,
          message: 'Details Box and Barcode Box overlap.',
        ),
      );
    }

    var requiredHeight = 0.0;

    for (final columnIndex in selectedColumns) {
      if (columnIndex >= excelData.headers.length ||
          columnIndex >= candidate.length) {
        continue;
      }

      final label = excelData.headers[columnIndex];
      final value = candidate[columnIndex];

      // Measure the label and value separately so the value can
      // wrap without being measured as one unbreakable line.
      final labelPainter = TextPainter(
        text: TextSpan(
          text: '$label : ',
          style: TextStyle(fontSize: fontSize, fontWeight: fontWeight),
        ),
        textDirection: TextDirection.ltr,
      )..layout();

      final valueWidth = availableWidth - labelPainter.width;

      if (valueWidth <= 0) {
        warnings.add(
          LayoutWarning(
            pageNumber: pageNumber,
            message: 'Field "$label" leaves no space for its value.',
          ),
        );
        continue;
      }

      final valuePainter = TextPainter(
        text: TextSpan(
          text: value,
          style: TextStyle(fontSize: fontSize, fontWeight: fontWeight),
        ),
        textDirection: TextDirection.ltr,
      )..layout(maxWidth: valueWidth);

      final lineHeight = valuePainter.height > labelPainter.height
          ? valuePainter.height
          : labelPainter.height;

      requiredHeight += lineHeight + configuration.fieldSpacing;
    }

    if (requiredHeight > availableHeight) {
      warnings.add(
        LayoutWarning(
          pageNumber: pageNumber,
          message:
              'Candidate details require '
              '${requiredHeight.toStringAsFixed(1)} pt, '
              'but only ${availableHeight.toStringAsFixed(1)} pt is available.',
        ),
      );
    }

    if (warnings.length >= 500) {
      warnings.add(
        const LayoutWarning(
          pageNumber: 1,
          message: 'Additional warnings were omitted.',
        ),
      );
      break;
    }
  }

  return warnings;
});

class A4PagePreview extends StatelessWidget {
  final LayoutConfiguration configuration;
  final List<MapEntry<String, String>> candidateDetails;
  final String barcodeValue;

  final ValueChanged<LayoutBoxType> onSelectBox;
  final BoxMoveCallback onMoveBox;
  final BoxResizeCallback onResizeBox;

  final bool zoomFirstHalf;

  const A4PagePreview({
    super.key,
    required this.configuration,
    required this.candidateDetails,
    required this.barcodeValue,
    required this.onSelectBox,
    required this.onMoveBox,
    required this.onResizeBox,
    this.zoomFirstHalf = false,
  });

  @override
  Widget build(BuildContext context) {
    return LayoutBuilder(
      builder: (context, constraints) {
        // In zoom mode, fit the top half of the A4 page
        // into the available preview area.
        final visiblePageHeightPoints = zoomFirstHalf
            ? LayoutConfiguration.pageHeight / 2
            : LayoutConfiguration.pageHeight;

        final widthScale = constraints.maxWidth / LayoutConfiguration.pageWidth;

        final heightScale = constraints.maxHeight / visiblePageHeightPoints;

        final scale = widthScale < heightScale ? widthScale : heightScale;

        final pageWidth = LayoutConfiguration.pageWidth * scale;
        final fullPageHeight = LayoutConfiguration.pageHeight * scale;

        final visiblePageHeight = visiblePageHeightPoints * scale;

        return Center(
          child: Focus(
            autofocus: true,
            onKeyEvent: _handleKeyEvent,
            child: Builder(
              builder: (focusContext) {
                return ClipRect(
                  child: SizedBox(
                    width: pageWidth,
                    height: visiblePageHeight,
                    child: OverflowBox(
                      alignment: Alignment.topCenter,
                      minWidth: pageWidth,
                      maxWidth: pageWidth,
                      minHeight: fullPageHeight,
                      maxHeight: fullPageHeight,
                      child: Container(
                        width: pageWidth,
                        height: fullPageHeight,
                        decoration: BoxDecoration(
                          color: Colors.white,
                          border: Border.all(color: Colors.grey.shade400),
                          boxShadow: [
                            BoxShadow(
                              color: Colors.black.withValues(alpha: 0.10),
                              blurRadius: 12,
                              offset: const Offset(0, 4),
                            ),
                          ],
                        ),
                        child: Stack(
                          children: [
                            Positioned(
                              left: 8 * scale,
                              top: 8 * scale,
                              child: Text(
                                zoomFirstHalf
                                    ? 'A4 PREVIEW · TOP HALF'
                                    : 'A4 PREVIEW',
                                style: TextStyle(
                                  color: Colors.grey.shade500,
                                  fontSize: 9 * scale,
                                ),
                              ),
                            ),

                            _buildPositionedBox(
                              context: focusContext,
                              box: configuration.detailsBox,
                              type: LayoutBoxType.details,
                              scale: scale,
                              selected:
                                  configuration.selectedBox ==
                                  LayoutBoxType.details,
                              child: _buildDetailsContent(scale),
                            ),

                            _buildPositionedBox(
                              context: focusContext,
                              box: configuration.barcodeBoxForPage(
                                    configuration.selectedPageNumber,
                                    ),
                              type: LayoutBoxType.barcode,
                              scale: scale,
                              selected:
                                  configuration.selectedBox ==
                                  LayoutBoxType.barcode,
                              child: _buildBarcodeContent(scale),
                            ),
                          ],
                        ),
                      ),
                    ),
                  ),
                );
              },
            ),
          ),
        );
      },
    );
  }

  KeyEventResult _handleKeyEvent(FocusNode node, KeyEvent event) {
    if (event is! KeyDownEvent && event is! KeyRepeatEvent) {
      return KeyEventResult.ignored;
    }

    final key = event.logicalKey;
    final isControlPressed = HardwareKeyboard.instance.isControlPressed;

    final selectedType = configuration.selectedBox;
    final step = configuration.movementStep;

    if (key != LogicalKeyboardKey.arrowUp &&
        key != LogicalKeyboardKey.arrowDown &&
        key != LogicalKeyboardKey.arrowLeft &&
        key != LogicalKeyboardKey.arrowRight) {
      return KeyEventResult.ignored;
    }

    if (isControlPressed) {
      switch (key) {
        case LogicalKeyboardKey.arrowUp:
          onResizeBox(selectedType, 0, -step);
          break;
        case LogicalKeyboardKey.arrowDown:
          onResizeBox(selectedType, 0, step);
          break;
        case LogicalKeyboardKey.arrowLeft:
          onResizeBox(selectedType, -step, 0);
          break;
        case LogicalKeyboardKey.arrowRight:
          onResizeBox(selectedType, step, 0);
          break;
      }
    } else {
      switch (key) {
        case LogicalKeyboardKey.arrowUp:
          onMoveBox(selectedType, 0, -step);
          break;
        case LogicalKeyboardKey.arrowDown:
          onMoveBox(selectedType, 0, step);
          break;
        case LogicalKeyboardKey.arrowLeft:
          onMoveBox(selectedType, -step, 0);
          break;
        case LogicalKeyboardKey.arrowRight:
          onMoveBox(selectedType, step, 0);
          break;
      }
    }

    return KeyEventResult.handled;
  }

  Widget _buildPositionedBox({
    required BuildContext context,
    required LayoutBox box,
    required LayoutBoxType type,
    required double scale,
    required bool selected,
    required Widget child,
  }) {
    return Positioned(
      left: box.x * scale,
      top: box.y * scale,
      width: box.width * scale,
      height: box.height * scale,
      child: GestureDetector(
        onTap: () {
          Focus.of(context).requestFocus();
          onSelectBox(type);
        },
        onPanStart: (_) {
          Focus.of(context).requestFocus();
          onSelectBox(type);
        },
        onPanUpdate: (details) {
          onMoveBox(type, details.delta.dx / scale, details.delta.dy / scale);
        },
        child: Container(
          padding: EdgeInsets.all(6 * scale),
          decoration: BoxDecoration(
            color: Colors.white,
            border: Border.all(
              color: selected ? Colors.blue : Colors.grey.shade500,
              width: selected ? 2 : 1,
            ),
          ),
          child: ClipRect(child: child),
        ),
      ),
    );
  }

  Widget _buildDetailsContent(double scale) {
    final isBold = configuration.textStyle == LayoutTextStyle.bold;

    if (candidateDetails.isEmpty) {
      return const Text(
        'No candidate data available',
        maxLines: 1,
        softWrap: false,
        style: TextStyle(color: Colors.grey),
      );
    }

    final fontSize = configuration.detailsFontSize * scale;

    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      mainAxisSize: MainAxisSize.min,
      children: [
        for (final field in candidateDetails)
          SizedBox(
            width: double.infinity,
            child: Text(
              '${field.key} : ${field.value}',
              maxLines: 1,
              softWrap: false,
              overflow: TextOverflow.clip,
              style: TextStyle(
                fontSize: fontSize,
                fontWeight: isBold ? FontWeight.bold : FontWeight.normal,
                color: Colors.black,
                height: 1.35,
              ),
            ),
          ),
      ],
    );
  }

  Widget _buildBarcodeContent(double scale) {
    const bars = <int>[
      2,
      1,
      3,
      1,
      1,
      2,
      2,
      1,
      3,
      1,
      2,
      2,
      1,
      3,
      1,
      2,
      1,
      1,
      3,
      2,
      2,
      1,
      1,
      3,
      2,
      1,
      2,
      3,
      1,
      1,
      2,
      2,
    ];

    return Column(
      mainAxisAlignment: MainAxisAlignment.center,
      children: [
        Expanded(
          child: Row(
            crossAxisAlignment: CrossAxisAlignment.stretch,
            children: [
              for (var i = 0; i < bars.length; i++)
                Expanded(
                  flex: bars[i],
                  child: Container(
                    color: i.isEven ? Colors.black : Colors.white,
                  ),
                ),
            ],
          ),
        ),
        if (configuration.showBarcodeNumber)
          Padding(
            padding: EdgeInsets.only(top: 4 * scale),
            child: Text(
              barcodeValue,
              maxLines: 1,
              softWrap: false,
              style: TextStyle(
                fontSize: 10 * scale,
                fontWeight: configuration.textStyle == LayoutTextStyle.bold
                    ? FontWeight.bold
                    : FontWeight.normal,
                color: Colors.black,
              ),
            ),
          ),
      ],
    );
  }
}
